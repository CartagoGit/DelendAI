import { readFile } from 'node:fs/promises';
import { checkSkillsManifest } from './snapshot/tools/scripts/lint/check-skills.script.ts';
import { packageSkillSource } from './snapshot/packages/core/src/lib/skills/sources/package-skill-source.ts';
const root = import.meta.dir;
const inventory = JSON.parse(await readFile(`${root}/inventory.json`, 'utf8'));
const manifest = JSON.parse(await readFile(`${root}/snapshot/packages/core/skills/manifest.json`, 'utf8'));
const issues = checkSkillsManifest(manifest, inventory.paths);
if (issues.length !== 1 || issues[0]?.kind !== 'missing-in-manifest' || !issues[0]?.detail.includes('read-stderr-storm')) throw Error('Unexpected manifest census');
console.log(JSON.stringify({case:'production-manifest-checker', snapshot:inventory.snapshot, discovered:inventory.count, manifest:manifest.skills.length, issues}));
const source = packageSkillSource({id:'audit-commit-policy', source:'plugin', packageRoot:`${root}/snapshot/plugins/commit-policy`, owner:'@delendai/commit-policy', packageVersion:'audit-fixed'});
const listed = await source.list();
const loaded = await source.load('delendai-read-stderr-storm');
if (listed.length !== 1 || loaded?.id !== 'delendai-read-stderr-storm') throw Error('Supplemental portable source did not find storm skill');
console.log(JSON.stringify({case:'portable-source-counterevidence', listed:listed.map(x=>x.id), loaded:loaded?.id, bodyMatches:loaded?.body === await readFile(`${root}/snapshot/plugins/commit-policy/skills/read-stderr-storm/SKILL.md`,'utf8')}));

// Identity-only check: real resolver, synthetic runtime catalog, no invoke method.
const { resolveIdentity } = await import('./snapshot/packages/core/src/lib/dispatch/capability-resolver.identity.ts');
const entry = {name:'delendai_proposals_proposals_close_plan', toolId:'proposals_close_plan', pluginId:'proposals'};
const runtime = {searchTools: () => [entry], getToolExposure: () => 'hidden', resolveRoute: () => undefined} as any;
const wrong = resolveIdentity(runtime, {qualifiedName:'delendai_proposals_close_plan'});
const qualified = resolveIdentity(runtime, {qualifiedName:entry.name});
const bare = resolveIdentity(runtime, {qualifiedName:entry.toolId});
if (wrong !== undefined || qualified?.kind !== 'resolved' || bare?.kind !== 'resolved') throw Error('Unexpected capability identity result');
console.log(JSON.stringify({case:'broker-identity-and-hidden-counterevidence', wrongRecipeQualifiedNameResolves:wrong !== undefined, correctQualifiedNameResolves:qualified.kind === 'resolved', bareRegisteredToolIdResolves:bare.kind === 'resolved', access:bare.kind === 'resolved' ? bare.identity.access : null, limits:'Real pure identity resolver with synthetic static-catalog port; does not invoke tool, activate plugin, or implement route metadata.'}));
