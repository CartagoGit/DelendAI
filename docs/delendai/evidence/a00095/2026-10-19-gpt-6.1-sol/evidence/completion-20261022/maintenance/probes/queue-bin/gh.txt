#!/usr/bin/python3
import os,sys,json,pathlib
args=sys.argv[1:]; mode=os.environ['AUDIT_CERT_MODE'];base='b'*40;head='a'*40
p=pathlib.Path(os.environ['AUDIT_QUEUE_CALLS']);s=p.with_suffix('.state');armed=s.exists()
with p.open('a') as f:f.write(json.dumps({'program':'owned-gh-port','args':args})+'\n')
def emit(v):print(json.dumps(v));sys.exit(0)
if args[0]=='pr' and '--auto' in args:s.touch();sys.exit(0)
if args[0]!='api':sys.exit('Blocked unmapped fake gh command')
route=args[1]
if '/pulls?' in route:emit([{'number':1,'title':'owned candidate','head':{'ref':'pub/fixture','sha':head},'base':{'ref':'integration'},'draft':False,'auto_merge':{} if armed else None}])
if route.endswith('/pulls/1'):emit({'mergeable_state':'clean'})
if '/check-runs?' in route:emit({'check_runs':[{'name':'required','conclusion':'success'}]})
if route.endswith('/commits/integration'):emit({'sha':base})
if '/workflows/ci.yml/runs?' in route:
 sha=base if 'head_sha='+base in route else head
 emit({'workflow_runs':[] if mode=='uncertified' else [{'head_sha':sha,'event':'workflow_dispatch','status':'in_progress' if mode=='pending' else 'completed','conclusion':None if mode=='pending' else 'failure' if mode=='red' else 'success'}]})
if '/compare/' in route:emit({'behind_by':0})
if '/actions/runs?' in route:emit({'workflow_runs':[]})
sys.exit('Blocked unmapped fake gh route '+route)
