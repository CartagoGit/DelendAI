import {mkdtemp,writeFile,readFile,access,rm} from 'node:fs/promises';import {join} from 'node:path';import {tmpdir} from 'node:os';
import {runAgentLockEngine} from '/tmp/delendai-audit-20261019-L2whP7/plugins/proposals/src/lib/locks/engine.ts';
const wait=async(p:string)=>{const until=Date.now()+15000;for(;;){try{await access(p);return;}catch{}if(Date.now()>until)throw Error('barrier timeout');await Bun.sleep(5);}};
if(process.argv[2]==='child'){
 const[dir,id,holder,stay]=process.argv.slice(3);await writeFile(join(dir,id+'.ready'),'');await wait(join(dir,'go'));
 const reply=await runAgentLockEngine({action:'claim',agent:id,task_id:id,files:['src/shared.ts'],...(holder==='agent'?{holder:'agent' as const}:{})},{lockPath:join(dir,'agents.lock.json')});
 await writeFile(join(dir,id+'.result'),JSON.stringify({pid:process.pid,reply:JSON.parse((reply.content[0] as any).text)}));
 if(stay==='live')await wait(join(dir,'release'));
}else{
 const rounds:any[]=[];const spawn=(dir:string,id:string,holder:string,stay:string)=>Bun.spawn(['bun',import.meta.path,'child',dir,id,holder,stay],{stdout:'pipe',stderr:'pipe',env:{PATH:process.env.PATH??''}});
 for(const holder of ['process','agent'])for(let round=0;round<50;round++){
  const dir=await mkdtemp(join(tmpdir(),'audit-g51-live-'));const a=spawn(dir,'A',holder,'live'),b=spawn(dir,'B',holder,'live');
  await Promise.all([wait(join(dir,'A.ready')),wait(join(dir,'B.ready'))]);await writeFile(join(dir,'go'),'');await Promise.all([wait(join(dir,'A.result')),wait(join(dir,'B.result'))]);
  const results=await Promise.all(['A','B'].map(async id=>JSON.parse(await readFile(join(dir,id+'.result'),'utf8'))));
  const alive=results.map(r=>{try{process.kill(r.pid,0);return true;}catch{return false;}});
  const final=JSON.parse(await readFile(join(dir,'agents.lock.json'),'utf8'));rounds.push({holder,round,alive,grants:results.filter(r=>r.reply.claimed===true).length,survivors:final.in_flight.map((r:any)=>r.task_id),results});
  await writeFile(join(dir,'release'),'');await Promise.all([a.exited,b.exited]);await rm(dir,{recursive:true,force:true});
 }
 const sequential:any[]=[];
 for(const holder of ['process','agent']){
  const dir=await mkdtemp(join(tmpdir(),'audit-g51-dead-'));await writeFile(join(dir,'go'),'');const a=spawn(dir,'A',holder,'exit');await a.exited;const b=spawn(dir,'B',holder,'exit');await b.exited;
  const results=await Promise.all(['A','B'].map(async id=>JSON.parse(await readFile(join(dir,id+'.result'),'utf8'))));const lock=JSON.parse(await readFile(join(dir,'agents.lock.json'),'utf8'));sequential.push({holder,results,survivors:lock.in_flight.map((r:any)=>r.task_id)});await rm(dir,{recursive:true,force:true});
 }
 const summary={liveRounds:rounds.length,bothAlive:rounds.every(r=>r.alive.every(Boolean)),doubleGrant:rounds.filter(r=>r.grants>1).length,incorrectWinnerCount:rounds.filter(r=>r.grants!==1||r.survivors.length!==1).length,sequential};
 await writeFile('/tmp/delendai-audit-completion-root/repro-g51-live.json',JSON.stringify({summary,rounds},null,2)+'\n');console.log(JSON.stringify(summary));
}
